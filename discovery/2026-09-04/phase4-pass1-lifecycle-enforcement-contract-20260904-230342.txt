   117	        dir=str(
   118	            path.parent

   120	        prefix=(
   121	            "."
   122	            + path.name

   125	        suffix=".tmp",

   128	    try:
   129	
   130	        with os.fdopen(
   131	            fd,
   132	            "w",

   144	            f.write("\n")

   146	            f.flush()

   152	        # Preserve the destination directory group.

   447	def build_policy_snapshot(
   448	    *,
   449	    config: PolicyConfig | None = None,
   450	) -> dict[str, Any]:
   451	
   452	    cfg = config or PolicyConfig()
   453	
   454	    tracker = _read_json(
   455	        TRACKER_PATH
   456	    )
   457	
   458	    records = tracker.get(
   459	        "records",
   460	        {},
   461	    )
   462	
   463	    if not isinstance(
   464	        records,
   465	        dict,
   466	    ):
   467	        raise ValueError(
   468	            "tracker records must be object"
   469	        )
   470	
   471	    decisions: list[
   472	        PolicyDecision
   473	    ] = []
   474	
   475	    counts = {
   476	        state.value: 0
   477	        for state
   478	        in PolicyState
   479	    }
   480	
   481	    publish_eligible = 0
   482	    quarantine_count = 0
   483	    deep_quarantine_count = 0
   484	    delete_candidate_shadow_count = 0
   485	
   486	
   487	    for config_id in sorted(
   488	        records
   489	    ):
   490	
   491	        record = records[
   492	            config_id
   493	        ]
   494	
   495	        decision = decide_policy(
   496	            record,
   497	            config=cfg,
   498	        )
   499	
   500	        decisions.append(
   501	            decision
   502	        )
   503	
   504	        counts[
   505	            decision.policy_state.value
   506	        ] += 1
   507	
   508	        publish_eligible += int(
   509	            decision.publish_eligible
   510	        )
   511	
   512	        quarantine_count += int(
   513	            decision.quarantine
   514	        )
   515	
   516	        deep_quarantine_count += int(
   517	            decision.deep_quarantine
   518	        )
   519	
   520	        delete_candidate_shadow_count += int(
   521	            decision.delete_candidate_shadow
   522	        )
   523	
   524	
   525	    payload = {
   526	        "schema_version": 1,
   527	
   528	        "generated_at":
   529	            now_iso(),
   530	
   531	        "mode":
   532	            "shadow",
   533	
   534	        "production_delete_enabled":
   535	            cfg.production_delete_enabled,
   536	
   537	        "policy": {
   538	            "deep_quarantine_after_unhealthy":
   539	                cfg.deep_quarantine_after_unhealthy,
   540	
   541	            "delete_candidate_after_unhealthy":
   542	                cfg.delete_candidate_after_unhealthy,
   543	
   544	            "healthy_recovery_immediate":
   545	                cfg.healthy_recovery_immediate,
   546	
   547	            "error_never_delete":
   548	                cfg.error_never_delete,
   549	        },
   550	
   551	        "tracked_count":
   552	            len(decisions),
   553	
   554	        "counts":
   555	            counts,
   556	
   557	        "publish_eligible":
   558	            publish_eligible,
   559	
   560	        "quarantine_count":
   561	            quarantine_count,
   562	
   563	        "deep_quarantine_count":
   564	            deep_quarantine_count,
   565	
   566	        "delete_candidate_shadow_count":
   567	            delete_candidate_shadow_count,
   568	
   569	        "production_delete_allowed_count":
   570	            0,
   571	
   572	        "decisions": [
   573	            {
   574	                **asdict(
   575	                    decision
   576	                ),
   577	                "policy_state":
   578	                    decision.policy_state.value,
   579	            }
   580	            for decision
   581	            in decisions
   582	        ],
   583	    }
   584	
   585	
   586	    write_performed = atomic_json_if_changed(
   587	        POLICY_STATE_PATH,
   588	        payload,
   589	    )
   590	
   591	    payload[
   592	        "write_performed"
   593	    ] = write_performed
   594	
   595	
   596	    return payload

   280	            )

   283	                now
   284	            )
   285	
   286	            return True, "state_changed"

   289	        if (
   290	            self._last_write_monotonic

   292	        ):
   293	
   294	            self._last_write_monotonic = (
   295	                now
   296	            )
