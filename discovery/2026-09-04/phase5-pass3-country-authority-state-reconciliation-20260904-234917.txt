   304	
   305	    elif source=="identity" and not locked:
   306	        reason="identity_not_locked"
   307	
   308	    elif not accepted:
   309	        reason=(
   310	            "state_not_accepted:"
   311	            + str(state)
   312	        )
   313	
   314	    else:
   315	        reason="accepted"
   316	
   317	
   318	    return CountryEvidence(
   319	        source=source,
   320	        config_id=config_id,
   321	        country_code=code,
   322	        country_name=name,
   323	        flag=clean(
   324	            obj.get("flag")
   325	        ),
   326	        confidence=confidence_of(
   327	            obj
   328	        ),
   329	        state=state,
   330	        locked=locked,
   331	        accepted=accepted,
   332	        reason=reason,
   333	    )

   422	def choose_projection(
   423	    config_id: str,
   424	    evidence: list[CountryEvidence],
   425	) -> CountryProjection:
   426	
   427	    accepted=[
   428	        row
   429	        for row in evidence
   430	        if row.accepted
   431	    ]
   432	
   433	    keys={
   434	        evidence_key(row)
   435	        for row in accepted
   436	        if evidence_key(row)
   437	    }
   438	
   439	
   440	    # Fail closed on conflicting accepted
   441	    # evidence.
   442	    if len(keys) > 1:
   443	
   444	        return CountryProjection(
   445	            config_id=config_id,
   446	            state="conflict",
   447	            country_code=None,
   448	            country_name=None,
   449	            flag=None,
   450	            confidence=None,
   451	            selected_source=None,
   452	            conflict=True,
   453	            evidence_count=len(evidence),
   454	            evidence=[
   455	                asdict(x)
   456	                for x in evidence
   457	            ],
   458	        )
   459	
   460	
   461	    priority={
   462	        "identity":0,
   463	        "results":1,
   464	        "pipeline_latest":2,
   465	    }
   466	
   467	
   468	    accepted.sort(
   469	        key=lambda row: (
   470	            priority.get(
   471	                row.source,
   472	                99,
   473	            ),
   474	            -(
   475	                row.confidence
   476	                if row.confidence
   477	                is not None
   478	                else -1
   479	            ),
   480	        )
   481	    )
   482	
   483	
   484	    if not accepted:
   485	
   486	        return CountryProjection(
   487	            config_id=config_id,
   488	            state="unknown",
   489	            country_code=None,
   490	            country_name=None,
   491	            flag=None,
   492	            confidence=None,
   493	            selected_source=None,
   494	            conflict=False,
   495	            evidence_count=len(evidence),
   496	            evidence=[
   497	                asdict(x)
   498	                for x in evidence
   499	            ],
   500	        )
   501	
   502	
   503	    selected=accepted[0]
   504	
   505	
   506	    return CountryProjection(
   507	        config_id=config_id,
   508	        state="resolved",
   509	        country_code=selected.country_code,
   510	        country_name=selected.country_name,
   511	        flag=selected.flag,
   512	        confidence=selected.confidence,
   513	        selected_source=selected.source,
   514	        conflict=False,
   515	        evidence_count=len(evidence),
   516	        evidence=[
   517	            asdict(x)
   518	            for x in evidence
   519	        ],
   520	    )
