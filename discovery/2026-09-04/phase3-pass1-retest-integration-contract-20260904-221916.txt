    29	def _qualified(
    30	    results: list[ProbeResult],
    31	) -> list[ProbeResult]:
    32	
    33	    qualified = []
    34	
    35	    for result in results:
    36	
    37	        if not result.success:
    38	            continue
    39	
    40	        # Real traffic must have been observed.
    41	        # For download, a provider such as
    42	        # Google generate_204 may legitimately
    43	        # return 0 bytes. Therefore success is
    44	        # trusted from the probe plugin itself.
    45	        #
    46	        # Upload plugins only set success=True
    47	        # after payload bytes were actually sent.
    48	        qualified.append(result)
    49	
    50	    return qualified
    51

    31	def is_retryable_result(
    32	    result: HealthResult,
    33	) -> bool:
    34	
    35	    if result.state not in {
    36	        HealthState.ERROR,
    37	        HealthState.RUNTIME_FAILED,
    38	    }:
    39	        return False
    40	
    41	    code = (
    42	        result.error_code
    43	        or ""
    44	    )
    45	
    46	    if code in RETRYABLE_ERROR_CODES:
    47	        return True
    48	
    49	    message = (
    50	        result.error_message
    51	        or ""
    52	    ).lower()
    53	
    54	    retry_markers = (
    55	        "sandboxporterror",
    56	        "port allocation",
    57	        "failed to allocate",
    58	        "address already in use",
    59	        "port unavailable",
    60	    )
    61	
    62	    return any(
    63	        marker in message
    64	        for marker in retry_markers
    65	    )

    68	def run_health_with_retry(
    69	    *,
    70	    config_id: str,
    71	    config_type: str,
    72	    source,
    73	    startup_timeout: float,
    74	    download_timeout: float,
    75	    upload_timeout: float,
    76	    upload_payload_bytes: int,
    77	    retry_count: int,
    78	    jitter_min: float = 0.05,
    79	    jitter_max: float = 0.35,
    80	    cancel_check=None,
    81	) -> RetryExecution:
    82	
    83	    retries_used = 0
    84	    retry_reason = None
    85	
    86	    for attempt in range(
    87	        retry_count + 1
    88	    ):
    89	
    90	        if (
    91	            cancel_check is not None
    92	            and cancel_check()
    93	        ):
    94	            raise InterruptedError(
    95	                "health_cancelled"
    96	            )
    97	
    98	        result = run_health_once(
    99	            config_id=config_id,
   100	            config_type=config_type,
   101	            source=source,
   102	
   103	            startup_timeout=(
   104	                startup_timeout
   105	            ),
   106	
   107	            download_timeout=(
   108	                download_timeout
   109	            ),
   110	
   111	            upload_timeout=(
   112	                upload_timeout
   113	            ),
   114	
   115	            upload_payload_bytes=(
   116	                upload_payload_bytes
   117	            ),
   118	
   119	            cancel_check=
   120	                cancel_check,
   121	        )
   122	
   123	        if not is_retryable_result(
   124	            result
   125	        ):
   126	            return RetryExecution(
   127	                result=result,
   128	                retries_used=retries_used,
   129	                retry_reason=retry_reason,
   130	            )
   131	
   132	        retry_reason = (
   133	            result.error_code
   134	            or "retryable_runtime_error"
   135	        )
   136	
   137	        if attempt >= retry_count:
   138	            return RetryExecution(
   139	                result=result,
   140	                retries_used=retries_used,
   141	                retry_reason=retry_reason,
   142	            )
   143	
   144	        retries_used += 1
   145	
   146	        delay = random.uniform(
   147	            jitter_min,
   148	            jitter_max,
   149	        )
   150	
   151	        deadline = (
   152	            time.monotonic()
   153	            + delay
   154	        )
   155	
   156	        while time.monotonic() < deadline:
   157	            if (
   158	                cancel_check is not None
   159	                and cancel_check()
   160	            ):
   161	                raise InterruptedError(
   162	                    "health_cancelled"
   163	                )
   164	
   165	            time.sleep(
   166	                min(
   167	                    0.05,
   168	                    max(
   169	                        0.0,
   170	                        deadline
   171	                        - time.monotonic()
   172	                    ),
   173	                )
   174	            )
   175	
   176	    raise RuntimeError(
   177	        "retry loop exhausted unexpectedly"
   178	    )

    59	    ) -> None:

    63	                "max_workers must be >= 1"

    66	        self.max_workers = max_workers
