   195	def create_signed_cookie(
   196	    token: str,
   197	    expires: int
   198	):
   199	    payload = (
   200	        f"{token}:{expires}"
   201	    )
   202	
   203	    signature = hmac.new(
   204	        SESSION_SECRET.encode(),
   205	        payload.encode(),
   206	        hashlib.sha256
   207	    ).hexdigest()
   208	
   209	    return (
   210	        f"{token}:"
   211	        f"{expires}:"
   212	        f"{signature}"
   213	    )

   216	def verify_signed_cookie(
   217	    value: str
   218	):
   219	    try:
   220	        token, expires_text, signature = (
   221	            value.split(":", 2)
   222	        )
   223	
   224	        expires = int(
   225	            expires_text
   226	        )
   227	
   228	        if time.time() > expires:
   229	            return None
   230	
   231	        payload = (
   232	            f"{token}:{expires}"
   233	        )
   234	
   235	        expected = hmac.new(
   236	            SESSION_SECRET.encode(),
   237	            payload.encode(),
   238	            hashlib.sha256
   239	        ).hexdigest()
   240	
   241	        if not hmac.compare_digest(
   242	            signature,
   243	            expected
   244	        ):
   245	            return None
   246	
   247	        return (
   248	            token,
   249	            expires
   250	        )
   251	
   252	    except Exception:
   253	        return None

   256	def is_authenticated(
   257	    request
   258	):
   259	    cleanup_sessions()
   260	
   261	    cookie = request.cookies.get(
   262	        SESSION_COOKIE
   263	    )
   264	
   265	    if not cookie:
   266	        return False
   267	
   268	    verified = verify_signed_cookie(
   269	        cookie
   270	    )
   271	
   272	    if not verified:
   273	        return False
   274	
   275	    token, expires = verified
   276	
   277	    if token not in sessions:
   278	        sessions[token] = {
   279	            "expires": expires
   280	        }
   281	
   282	    return True
