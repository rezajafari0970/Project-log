   274	def healthy_candidates() -> list[
   275	    tuple[int,str,dict,dict]
   276	]:
   277	
   278	    now=now_epoch()
   279	
   280	    state=load_worker_state()
   281	
   282	    worker_records=state[
   283	        "records"
   284	    ]
   285	
   286	    result=[]
   287	
   288	
   289	    for hp in HEALTH_ROOT.glob(
   290	        "*.json"
   291	    ):
   292	
   293	        health=load_json(
   294	            hp
   295	        )
   296	
   297	        if not health:
   298	            continue
   299	
   300	        if str(
   301	            health.get(
   302	                "state",
   303	                "",
   304	            )
   305	        ).lower() != "healthy":
   306	            continue
   307	
   308	
   309	        config_id=hp.stem
   310	
   311	        cp=(
   312	            CONFIG_ROOT
   313	            / f"{config_id}.json"
   314	        )
   315	
   316	        record=load_json(
   317	            cp
   318	        )
   319	
   320	        if not record:
   321	            continue
   322	
   323	
   324	        cstate=country_state(
   325	            config_id
   326	        )
   327	
   328	
   329	        previous=worker_records.get(
   330	            config_id,
   331	            {}
   332	        )
   333	
   334	        last_run=int(
   335	            previous.get(
   336	                "last_run_epoch",
   337	                0,
   338	            )
   339	            or 0
   340	        )
   341	
   342	
   343	        retry=RETRY_SECONDS.get(
   344	            cstate,
   345	            3 * 60,
   346	        )
   347	
   348	        # Country is a one-time classification.
   349	        #
   350	        # Once a country verdict is confirmed,
   351	        # this config_id permanently leaves the
   352	        # Country scheduler. Periodic Health tests
   353	        # continue independently.
   354	        if retry is None:
   355	            continue
   356	
   357	        due=(
   358	            last_run
   359	            + retry
   360	        )
   361	
   362	        if now < due:
   363	            continue
   364	
   365	
   366	        # Lower number = higher priority.
   367	        priority={
   368	            "missing":0,
   369	            "unknown":1,
   370	            "ambiguous":1,
   371	            "unstable_exit":1,
   372	            "error":2,
   373	            "pending_confirmation":3,
   374	            "rotating":4,
   375	        }.get(
   376	            cstate,
   377	            3,
   378	        )
   379	
   380	
   381	        result.append(
   382	            (
   383	                priority,
   384	                config_id,
   385	                record,
   386	                health,
   387	            )
   388	        )
   389	
   390	
   391	    result.sort(
   392	        key=lambda x:(
   393	            x[0],
   394	            x[1],
   395	        )
   396	    )
   397	
   398	    return result

   635	def run_country_worker_adaptive(
   636	    *,
   637	    max_jobs: int,
   638	    concurrency: int,
   639	):
   640	    """
   641	    Adaptive parallel Country batch.
   642	
   643	    Selection and global ownership remain protected by
   644	    the Country Worker lock.
   645	
   646	    Each config owns an independent Xray sandbox.
   647	    """
   648	
   649	    from concurrent.futures import (
   650	        ThreadPoolExecutor,
   651	        as_completed,
   652	    )
   653	
   654	    concurrency=max(
   655	        1,
   656	        min(
   657	            int(concurrency),
   658	            8,
   659	        ),
   660	    )
   661	
   662	    max_jobs=max(
   663	        1,
   664	        min(
   665	            int(max_jobs),
   666	            24,
   667	        ),
   668	    )
   669	
   670	
   671	    WORKER_ROOT.mkdir(
   672	        parents=True,
   673	        exist_ok=True,

   678	
   679	        started=utc_now()
   680

   682	            load_worker_state()
   683	        )
   684	
   685	        records=worker_state[
   686	            "records"

   691	            healthy_candidates()
   692	        )

   694	        selected=candidates[
   695	            :max_jobs
   696	        ]

   724	                        config_id,

   727	                        "error",

   730	                        "adaptive_worker_exception",

   733	                        str(e)[:1000],

   331	def _process_event(
   332	    event: dict,

   338	        )

   341	
   342	    config_id=str(
   343	        event.get(
   344	            "config_id"

   346	        or ""

   348	
   349	
   350	    if (
   351	        not event_id
   352	        or not config_id
   353	    ):
   354	        raise ValueError(
   355	            "event missing event_id/config_id"
   356	        )
   357	
   358	
   359	    fast,fast_source=(
   360	        _reusable_fast(

   373	    exit_ip=str(
   374	        fast["exit_ip"]

   376	
   377	
   378	    # K6C:
   379	    # Country detection is performed at most once
   380	    # after successful consensus for this config.
   381	    #
   382	    # Future Health generations may have a new
   383	    # Exit-IP, but Geo providers are NOT called
   384	    # again for a config with locked Country.
   385	    identity=load_identity(
   386	        config_id

   615	def worker_loop(
   616	    worker_id: int,
   617	    retire_event: threading.Event | None = None,
   618	) -> None:
   619	
   620	    owner=(
   621	        f"country-event-consumer:"
   622	        f"{os.getpid()}:"
   623	        f"{worker_id}"
   624	    )
   625	
   626	
   627	    # FIX22_K8_REAL_WORKER_RETIREMENT
   628	    #
   629	    # Retirement is checked only before acquiring
   630	    # the next lease. A worker already processing an
   631	    # event always finishes ACK/NACK first.
   632	    while not STOP.is_set():
   633	
   634	        if (
   635	            retire_event is not None
   636	            and retire_event.is_set()
   637	        ):
   638	            _metric(
   639	                {
   640	                    "status":"worker_retired",
   641	                    "worker_id":worker_id,
   642	                    "reason":"controller_scale_down",
   643	                }
   644	            )
   645	            return
   646	
   647	        lease_path=None
   648	        event=None
   649	
   650	        try:
   651	
   652	            recover_expired()
   653	
   654	            leased=lease(
   655	                owner=owner,
   656	                seconds=90,
   657	            )
   658	
   659	
   660	            if leased is None:
   661	
   662	                if (
   663	                    retire_event is not None
   664	                    and retire_event.wait(0.20)
   665	                ):
   666	                    _metric(
   667	                        {
   668	                            "status":"worker_retired",
   669	                            "worker_id":worker_id,
   670	                            "reason":"idle_retirement",
   671	                        }
   672	                    )
   673	                    return
   674	
   675	                if STOP.is_set():
   676	                    return
   677	
   678	                continue
   679	
   680	
   681	            lease_path,event=leased
   682	
   683	
   684	            _metric(
   685	                {
   686	                    "status":
   687	                        "leased",
   688	
   689	                    "worker_id":
   690	                        worker_id,
   691	
   692	                    "event_id":
   693	                        event.get(
   694	                            "event_id"
   695	                        ),
   696	
   697	                    "config_id":
   698	                        event.get(
   699	                            "config_id"
   700	                        ),
   701	                }
   702	            )
   703	
   704	
   705	            result=_process_event(
   706	                event
   707	            )
   708	
   709	
   710	            ack(
   711	                lease_path,
   712	                result={
   713	                    "consumer":
   714	                        "k1-event-consumer",
   715	
   716	                    "processed":
   717	                        True,
   718	
   719	                    "country_code":
   720	                        result.get(
   721	                            "country_code"
   722	                        ),
   723	
   724	                    "state":
   725	                        result.get(
   726	                            "state"
   727	                        ),
   728	                },
   729	            )
   730	
   731	
   732	            _metric(
   733	                {
   734	                    "status":
   735	                        "acked",
   736	
   737	                    "worker_id":
   738	                        worker_id,
   739	
   740	                    "event_id":
   741	                        event.get(
   742	                            "event_id"
   743	                        ),
   744	
   745	                    "config_id":
   746	                        event.get(
   747	                            "config_id"
   748	                        ),
   749	                }
   750	            )
   751	
   752	
   753	        except MissingReusableExit as exc:
   754	
   755	            _metric(
   756	                {
   757	                    "status":
   758	                        "deferred_no_exit",
   759	
   760	                    "worker_id":
   761	                        worker_id,
   762	
   763	                    "event_id":
   764	                        (
   765	                            event.get(
   766	                                "event_id"
   767	                            )
   768	                            if isinstance(
   769	                                event,
   770	                                dict,
   771	                            )
   772	                            else None
   773	                        ),
   774	
   775	                    "config_id":
   776	                        (
   777	                            event.get(
   778	                                "config_id"
   779	                            )
   780	                            if isinstance(
   781	                                event,
   782	                                dict,
   783	                            )
   784	                            else None
   785	                        ),
   786	                }
   787	            )
   788	
   789	
   790	            if lease_path is not None:
   791	
   792	                try:
   793	
   794	                    # Old pre-K2 backlog should not
   795	                    # starve newer usable events.
   796	                    nack(
   797	                        lease_path,
   798	                        str(exc),
   799	                        retry_seconds=300,
   800	                        max_attempts=100,
   801	                    )
   802	
   803	                except Exception as nack_exc:
   804	
   805	                    _metric(
   806	                        {
   807	                            "status":
   808	                                "nack_error",
   809	
   810	                            "error":
   811	                                (
   812	                                    f"{type(nack_exc).__name__}: "
   813	                                    f"{nack_exc}"
   814	                                )[:1000],
   815	                        }
   816	                    )
   817	
   818	
   819	        except Exception as exc:
   820	
   821	            error=(
   822	                f"{type(exc).__name__}: "
   823	                f"{exc}"
   824	            )[:1000]
   825	
   826	
   827	            _metric(
   828	                {
   829	                    "status":
   830	                        "error",
   831

   129	def clean(
   130	    value: Any,
   131	) -> str | None:
   132	
   133	    if value is None:
   134	        return None
   135	
   136	    if not isinstance(
   137	        value,
   138	        (
   139	            str,
   140	            int,
   141	            float,
   142	        ),
   143	    ):
   144	        return None
   145	
   146	    value=str(value).strip()
   147	
   148	    if value in UNKNOWN_VALUES:
   149	        return None
   150	
   151	    return value or None
   152

   154	def normalize_code(
   155	    value: Any,
   156	) -> str | None:
   157	
   158	    value=clean(value)
   159	
   160	    if not value:
   161	        return None
   162	
   163	    value=value.upper()
   164	
   165	    if (
   166	        len(value)==2
   167	        and value.isalpha()
   168	    ):
   169	        return value
   170	
   171	    return None
   172

   174	def normalize_name(
   175	    value: Any,
   176	) -> str | None:
   177	
   178	    value=clean(value)
   179	
   180	    if not value:
   181	        return None
   182	
   183	    if (
   184	        len(value)==2
   185	        and value.isalpha()
   186	    ):
   187	        return None
   188	
   189	    return value

   192	def normalize_confidence(
   193	    value: Any,
   194	) -> float | None:
   195	
   196	    try:
   197	        value=float(value)
   198	
   199	    except Exception:
   200	        return None
   201	
   202	    if value > 1:
   203	        value=value / 100.0
   204	
   205	    return max(
   206	        0.0,
   207	        min(
   208	            1.0,
   209	            value,
   210	        ),
   211	    )

   214	def state_value(
   215	    obj: dict[str,Any],
   216	) -> str | None:
   217	
   218	    for key in (
   219	        "state",
   220	        "country_state",
   221	        "status",
   222	        "verdict",
   223	        "decision",
   224	        "resolution_state",
   225	    ):
   226	
   227	        value=clean(
   228	            obj.get(key)
   229	        )
   230	
   231	        if value:
   232	            return value.lower()
   233	
   234	    return None

   237	def confidence_value(
   238	    obj: dict[str,Any],
   239	) -> float | None:
   240	
   241	    for key in (
   242	        "country_confidence",
   243	        "confidence",
   244	        "score",
   245	    ):
   246	
   247	        if key in obj:
   248	
   249	            value=normalize_confidence(
   250	                obj.get(key)
   251	            )
   252	
   253	            if value is not None:
   254	                return value
   255	
   256	    return None

   259	def direct_country(
   260	    obj: dict[str,Any],
   261	) -> tuple[
   262	    str | None,
   263	    str | None,
   264	    str | None,
   265	]:
   266	
   267	    code=normalize_code(
   268	        obj.get(
   269	            "country_code"
   270	        )
   271	    )
   272	
   273	    name=normalize_name(
   274	        obj.get(
   275	            "country_name"
   276	        )
   277	    )
   278	
   279	
   280	    country=clean(
   281	        obj.get(
   282	            "country"
   283	        )
   284	    )
   285	
   286	    if country:
   287	
   288	        if (
   289	            not code
   290	            and len(country)==2
   291	            and country.isalpha()
   292	        ):
   293	
   294	            code=country.upper()
   295	
   296	        elif not name:
   297	
   298	            name=normalize_name(
   299	                country
   300	            )
   301	
   302	
   303	    flag=clean(
   304	        obj.get("flag")
   305	    )
   306	
   307	
   308	    return (
   309	        code,
   310	        name,
   311	        flag,
   312	    )

   315	def identity_locked(
   316	    obj: dict[str,Any],
   317	) -> bool:
   318	
   319	    metadata=(
   320	        obj.get("metadata")
   321	        if isinstance(
   322	            obj.get("metadata"),
   323	            dict,
   324	        )
   325	        else {}
   326	    )
   327	
   328	
   329	    values=[
   330	        obj.get(
   331	            "country_identity_locked"
   332	        ),
   333	
   334	        obj.get("locked"),
   335	
   336	        metadata.get(
   337	            "country_identity_locked"
   338	        ),
   339	    ]
   340	
   341	
   342	    if any(
   343	        value is True
   344	        for value in values
   345	    ):
   346	        return True
   347	
   348	
   349	    guard=(
   350	        obj.get(
   351	            "country_identity_guard"
   352	        )
   353	        or metadata.get(
   354	            "country_identity_guard"
   355	        )
   356	    )
   357	
   358	
   359	    if guard is True:
   360	        return True
   361	
   362	
   363	    if str(
   364	        guard
   365	    ).strip().lower() in {
   366	        "locked",
   367	        "true",
   368	        "1",
   369	    }:
   370	        return True
   371	
   372	
   373	    return False

   457	def result_evidence(
   458	    config_id: str,
   459	    obj: dict[str,Any],
   460	) -> list[CountryEvidence]:
   461	
   462	    rows=[]
   463	
   464	    top_state=state_value(
   465	        obj
   466	    )
   467	
   468	
   469	    # Canonical final result:
   470	    # top-level result is accepted only
   471	    # when result state itself is final.
   472	    top_accepted=(
   473	        top_state
   474	        in FINAL_STATES
   475	    )
   476	
   477	
   478	    row=section_evidence(
   479	        source="results",
   480	        config_id=config_id,
   481	        obj=obj,
   482	        section_name="$",
   483	        accepted=top_accepted,
   484	        state=top_state,
   485	        reason=(
   486	            "final_result"
   487	            if top_accepted
   488	            else (
   489	                "non_final_result:"
   490	                + str(top_state)
   491	            )
   492	        ),
   493	    )
   494	
   495	    if row:
   496	        rows.append(row)
   497	
   498	
   499	    # Primary is evidence, but does not
   500	    # override a non-final top-level result.
   501	    primary=obj.get(
   502	        "primary"
   503	    )
   504	
   505	    if isinstance(
   506	        primary,
   507	        dict,
   508	    ):
   509	
   510	        primary_state=state_value(
   511	            primary
   512	        )
   513	
   514	        primary_final=(
   515	            primary_state
   516	            in FINAL_STATES
   517	        )
   518	
   519	        accepted=(
   520	            top_accepted
   521	            and primary_final
   522	        )
   523	
   524	        row=section_evidence(
   525	            source="results",
   526	            config_id=config_id,
   527	            obj=primary,
   528	            section_name="$.primary",
   529	            accepted=accepted,
   530	            state=primary_state,
   531	            reason=(
   532	                "final_result_primary"
   533	                if accepted
   534	                else "primary_not_canonical_final"
   535	            ),
   536	        )
   537	
   538	        if row:
   539	            rows.append(row)
   540	
   541	
   542	    return rows

   545	def pipeline_evidence(
   546	    config_id: str,
   547	    obj: dict[str,Any],
   548	) -> list[CountryEvidence]:
   549	
   550	    rows=[]
   551	
   552	    top_state=state_value(
   553	        obj
   554	    )
   555	
   556	
   557	    # Hard boundary:
   558	    # non-final pipeline state may contain
   559	    # Country evidence, but it remains
   560	    # shadow-only and cannot resolve.
   561	    top_final=(
   562	        top_state
   563	        in FINAL_STATES
   564	    )
   565	
   566	
   567	    row=section_evidence(
   568	        source="pipeline_latest",
   569	        config_id=config_id,
   570	        obj=obj,
   571	        section_name="$",
   572	        accepted=top_final,
   573	        state=top_state,
   574	        reason=(
   575	            "final_pipeline"
   576	            if top_final
   577	            else (
   578	                "non_final_pipeline:"
   579	                + str(top_state)
   580	            )
   581	        ),
   582	    )
   583	
   584	    if row:
   585	        rows.append(row)
   586	
   587	
   588	    # Fusion is the strongest nested verdict
   589	    # when the parent pipeline state is final.
   590	    fusion=obj.get(
   591	        "fusion"
   592	    )
   593	
   594	    if isinstance(
   595	        fusion,
   596	        dict,
   597	    ):
   598	
   599	        fusion_state=state_value(
   600	            fusion
   601	        )
   602	
   603	        accepted=(
   604	            top_final
   605	            and fusion_state
   606	            in FINAL_STATES
   607	        )
   608	
   609	        row=section_evidence(
   610	            source="pipeline_latest",
   611	            config_id=config_id,
   612	            obj=fusion,
   613	            section_name="$.fusion",
   614	            accepted=accepted,
   615	            state=fusion_state,
   616	            reason=(
   617	                "final_pipeline_fusion"
   618	                if accepted
   619	                else "fusion_not_canonical_final"
   620	            ),
   621	        )
   622	
   623	        if row:
   624	            rows.append(row)
   625	
   626	
   627	    # Primary is usable evidence only when
   628	    # both parent and primary are final.
   629	    primary=obj.get(
   630	        "primary"
   631	    )
   632	
   633	    if isinstance(
   634	        primary,
   635	        dict,
   636	    ):
   637	
   638	        primary_state=state_value(
   639	            primary
   640	        )
   641	
   642	        accepted=(
   643	            top_final
   644	            and primary_state
   645	            in FINAL_STATES
   646	        )
   647	
   648	        row=section_evidence(
   649	            source="pipeline_latest",
   650	            config_id=config_id,
   651	            obj=primary,
   652	            section_name="$.primary",
   653	            accepted=accepted,
   654	            state=primary_state,
   655	            reason=(
   656	                "final_pipeline_primary"
   657	                if accepted
   658	                else "primary_not_canonical_final"
   659	            ),
   660	        )
   661	
   662	        if row:
   663	            rows.append(row)
   664	
   665	
   666	    return rows

   693

   714

   720	
   721	            out.extend(
   722	                result_evidence(

   778	def accepted_source_rows(
   779	    evidence: list[CountryEvidence],
   780	) -> dict[
   781	    str,
   782	    list[CountryEvidence],
   783	]:
   784	
   785	    grouped={}
   786	
   787	    for row in evidence:
   788	
   789	        if not row.accepted:
   790	            continue
   791	
   792	        grouped.setdefault(
   793	            row.source,
   794	            []
   795	        ).append(row)
   796	
   797	    return grouped

   800	def source_consensus(
   801	    rows: list[CountryEvidence],
   802	) -> CountryEvidence | None:
   803	
   804	    if not rows:
   805	        return None
   806	
   807	
   808	    keys={
   809	        evidence_key(row)
   810	        for row in rows
   811	        if evidence_key(row)
   812	    }
   813	
   814	
   815	    if len(keys) != 1:
   816	        return None
   817	
   818	
   819	    # Prefer strongest nested path.
   820	    path_priority={
   821	        "$.fusion":0,
   822	        "$":1,
   823	        "$.primary":2,
   824	    }
   825	
   826	
   827	    return sorted(
   828	        rows,
   829	        key=lambda row: (
   830	            path_priority.get(
   831	                row.evidence_path
   832	                or "",
   833	                99,
   834	            ),
   835	
   836	            -(
   837	                row.confidence
   838	                if row.confidence
   839	                is not None
   840	                else -1
   841	            ),
   842	        ),
   843	    )[0]

   846	def choose_projection(
   847	    config_id: str,
   848	    evidence: list[CountryEvidence],
   849	) -> CountryProjection:
   850	
   851	    grouped=accepted_source_rows(
   852	        evidence
   853	    )
   854	
   855	
   856	    candidates={}
   857	
   858	
   859	    for source,rows in grouped.items():
   860	
   861	        row=source_consensus(
   862	            rows
   863	        )
   864	
   865	        if row is not None:
   866	
   867	            candidates[
   868	                source
   869	            ]=row
   870	
   871	
   872	    # Different accepted sources must agree.
   873	    keys={
   874	        evidence_key(row)
   875	        for row in candidates.values()
   876	        if evidence_key(row)
   877	    }
   878	
   879	
   880	    if len(keys) > 1:
   881	
   882	        return CountryProjection(
   883	            config_id=config_id,
   884	
   885	            state="conflict",
   886	
   887	            country_code=None,
   888	            country_name=None,
   889	            flag=None,
   890	
   891	            confidence=None,
   892	
   893	            selected_source=None,
   894	            selected_path=None,
   895	
   896	            conflict=True,
   897	
   898	            evidence_count=len(
   899	                evidence
   900	            ),
   901	
   902	            accepted_evidence_count=sum(
   903	                1
   904	                for row in evidence
   905	                if row.accepted
   906	            ),
   907	
   908	            evidence=[
   909	                asdict(row)
   910	                for row in evidence
   911	            ],
   912	        )
   913	
   914	
   915	    authority=(
   916	        "identity",
   917	        "results",
   918	        "pipeline_latest",
   919	    )
   920	
   921	
   922	    selected=None
   923	
   924	
   925	    for source in authority:
   926	
   927	        if source in candidates:
   928	
   929	            selected=candidates[
   930	                source
   931	            ]
   932	
   933	            break
   934	
   935	
   936	    if selected is None:
   937	
   938	        return CountryProjection(
   939	            config_id=config_id,
   940

   185	def select_targets(
   186	    snapshot: PublishSnapshot,
   187	    projection: dict[str, Any],
   188	) -> tuple[str, str]:
   189	
   190	    publish_ids = {
   191	        str(record["id"])
   192	        for record in snapshot.configs
   193	        if (
   194	            isinstance(record, dict)
   195	            and record.get("id")
   196	        )
   197	    }
   198	
   199	    country_counts: dict[str, int] = {}
   200	
   201	    for cid, row in projection.get(
   202	        "records",
   203	        {},
   204	    ).items():
   205	
   206	        if str(cid) not in publish_ids:
   207	            continue
   208	
   209	        if row.get(
   210	            "state"
   211	        ) != "resolved":
   212	            continue
   213	
   214	        code = row.get(
   215	            "country_code"
   216	        )
   217	
   218	        if not isinstance(
   219	            code,
   220	            str,
   221	        ):
   222	            continue
   223	
   224	        code = code.strip().upper()
   225	
   226	        if (
   227	            len(code) != 2
   228	            or not code.isalpha()
   229	        ):
   230	            continue
   231	
   232	        country_counts[code] = (
   233	            country_counts.get(
   234	                code,
   235	                0,
   236	            )
   237	            + 1
   238	        )
   239	
   240	    if not country_counts:
   241	        raise RuntimeError(
   242	            "no resolved publishable country"
   243	        )
   244	
   245	    country = max(
   246	        country_counts,
   247	        key=country_counts.get,
   248	    )
   249	
   250	    type_counts: dict[str, int] = {}
   251	
   252	    for record in snapshot.configs:
   253	
   254	        if not isinstance(
   255	            record,
   256	            dict,
   257	        ):
   258	            continue
   259	
   260	        config_type = record.get(
   261	            "type"
   262	        )
   263	
   264	        if not isinstance(
   265	            config_type,
   266	            str,
   267	        ):
   268	            continue
   269	
   270	        config_type = (
   271	            config_type
   272	            .strip()
   273	            .lower()
   274	        )
   275	
   276	        if not config_type:
   277	            continue
   278	
   279	        type_counts[
   280	            config_type
   281	        ] = (
   282	            type_counts.get(
   283	                config_type,
   284	                0,
   285	            )
   286	            + 1
   287	        )
   288	
   289	    if not type_counts:
   290	        raise RuntimeError(
   291	            "no publishable config type"
   292	        )
   293	
   294	    config_type = max(
   295	        type_counts,
   296	        key=type_counts.get,

   299	    return (
   300	        country,
   301	        config_type,

   130	            )

   152	                dict,

   156	                "healthy"
   157	            ) is True
   158	            and

   433	def nack(
   434	    lease_path: Path,
   435	    error: str,

   437	    retry_seconds: int=10,

   446	                "lease_missing"
   447	            )

   450	            o.get("attempt",0)
   451	        )+1
   452	
   453	        o["attempt"]=attempt
   454	        o["last_error"]=(
   455	            str(error)[:1000]

   458	        o.pop(
   459	            "lease_owner",
   460	            None,
   461	        )

   463	            "lease_until_epoch",
   464	            None,

   471	                k+".json"

   474	        else:
   475	            delay=min(
   476	                900,
   477	                max(
   478	                    1,
   479	                    int(retry_seconds)
   480	                )
   481	                *(
   482	                    2 ** min(
   483	                        attempt-1,
   484	                        5,
   485	                    )
   486	                ),
   487	            )

   489	            o["not_before_epoch"]=(
   490	                now()+delay
   491	            )

    91	def observe(
    92	    *,
    93	    config_id: str,
    94	    exit_ip: str,
    95	    country_code: str | None,
    96	    observed_epoch: int | None = None,
    97	    minimum_repeat_seconds: int = MIN_REPEAT_SECONDS,
    98	    max_history: int = MAX_HISTORY,
    99	) -> dict:
   100	
   101	    now=int(
   102	        observed_epoch
   103	        if observed_epoch is not None
   104	        else time.time()
   105	    )
   106	
   107	    state=_read(config_id)
   108	
   109	    obs=list(
   110	        state.get(
   111	            "observations",
   112	            []
   113	        )
   114	    )
   115	
   116	    code=(
   117	        str(country_code)
   118	        .strip()
   119	        .upper()
   120	        if country_code
   121	        else None
   122	    )
   123	
   124	    row={
   125	        "ts":now,
   126	        "exit_ip":str(exit_ip),
   127	        "country_code":code,
   128	    }
   129	
   130	
   131	    append=True
   132	
   133	    if obs:
   134	
   135	        last=obs[-1]
   136	
   137	        same=(
   138	            last.get("exit_ip")
   139	            ==row["exit_ip"]
   140	            and last.get("country_code")
   141	            ==row["country_code"]
   142	        )
   143	
   144	        age=(
   145	            now
   146	            -int(
   147	                last.get("ts",0)
   148	            )
   149	        )
   150	
   151	        # Same observation in the same immediate
   152	        # execution is noise, but the same IP/country
   153	        # observed again in a later Health generation
   154	        # is real temporal evidence.
   155	        if (
   156	            same
   157	            and age
   158	            <max(
   159	                1,
   160	                int(
   161	                    minimum_repeat_seconds
   162	                )
   163	            )
   164	        ):
   165	            append=False
   166	
   167	
   168	    if append:
   169	        obs.append(row)
   170	
   171	
   172	    obs=obs[
   173	        -max(
   174	            3,
   175	            int(max_history),
   176	        ):
   177	    ]
   178	
   179	    state["observations"]=obs
   180	
   181	
   182	    ips=[
   183	        x.get("exit_ip")
   184	        for x in obs
   185	        if x.get("exit_ip")
   186	    ]
   187	
   188	    countries=[
   189	        str(
   190	            x.get("country_code")
   191	        ).upper()
   192	        for x in obs
   193	        if x.get("country_code")
   194	    ]
   195	
   196	    unique_ips=set(ips)
   197	    unique_countries=set(countries)
   198	
   199	    counts=Counter(countries)
   200	
   201	    dominant_country=None
   202	    dominant_count=0
   203	
   204	    if counts:
   205	
   206	        dominant_country,dominant_count=(
   207	            counts.most_common(1)[0]
   208	        )
   209	
   210	
   211	    verdict="pending_confirmation"
   212	    reason="insufficient_history"
   213	
   214	
   215	    if len(obs)>=2:
   216	
   217	        # Stable endpoint:
   218	        # same country and same IP observed in
   219	        # separate temporal observations.
   220	        if (
   221	            len(unique_countries)==1
   222	            and len(unique_ips)==1
   223	        ):
   224	
   225	            verdict="confirmed_stable"
   226	            reason="same_ip_same_country_temporal"
   227	
   228	
   229	        # Rotating endpoint but geo-stable.
   230	        elif (
   231	            len(unique_countries)==1
   232	            and len(unique_ips)>=2
   233	        ):
   234	
   235	            verdict="confirmed_rotating_ip"
   236	            reason="rotating_ip_same_country"
   237	
   238	
   239	        elif len(unique_countries)>=2:
   240	
   241	            # Do not flip the final country because
   242	            # one provider/exit observation disagrees.
   243	            #
   244	            # A dominant 2/3+ majority keeps the
   245	            # dominant country but requests recovery.
   246	            if (
   247	                len(countries)>=3
   248	                and dominant_count>=2
   249	                and (
   250	                    dominant_count
   251	                    /len(countries)
   252	                )>=0.66
   253	            ):
   254	
   255	                verdict="pending_confirmation"
   256	                reason="country_change_majority_recovery"
   257	
   258	            else:
   259	
   260	                verdict="pending_confirmation"
   261	                reason="country_change_ambiguous"
   262	
   263	
   264	    result={
   265	        "state":verdict,
   266	        "reason":reason,
   267	
   268	        "observation_count":
   269	            len(obs),
   270	
   271	        "unique_ip_count":
   272	            len(unique_ips),
   273	
   274	        "unique_country_count":
   275	            len(unique_countries),
   276	
   277	        "dominant_country":
   278	            dominant_country,
   279	
   280	        "dominant_country_count":
   281	            dominant_count,
   282	
   283	        "observation_appended":
   284	            append,
   285	
   286	        "observations":
   287	            obs,
   288	    }
   289	
   290	
   291	    state["last_verdict"]=result
   292	
   293	    _write(
   294	        config_id,
   295	        state,
   296	    )
   297	
   298	    return result
